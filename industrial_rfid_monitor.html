Ejemplo de código JavaScript para escuchar lecturas en tiempo real con Firebase Firestore:
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.x.x/firebase-app.js";
import { getFirestore, collection, onSnapshot, query, orderBy, limit } from "https://www.gstatic.com/firebasejs/10.x.x/firebase-firestore.js";

const firebaseConfig = { apiKey: "TU_API_KEY", projectId: "industria-rfid" };
const app = initializeApp(firebaseConfig);
const db = getFirestore(app);

// Escuchar lecturas RFID en tiempo real
const q = query(collection(db, "lecturas_rfid"), orderBy("timestamp", "desc"), limit(10));
onSnapshot(q, (snapshot) => {
    snapshot.docChanges().forEach((change) => {
        if (change.type === "added") {
            const data = change.doc.data();
            console.log("Nueva lectura RFID detectada:", data.epc, data.antena);
            // Actualizar interfaz web automáticamente
        }
    });
});
